-- 1 ---------------------------------------------------------------- parts: buying price
alter table public.part_price_guide add column if not exists buy_price numeric;
alter table public.part_price_guide drop constraint if exists part_price_guide_buy_price_check;
alter table public.part_price_guide add constraint part_price_guide_buy_price_check
  check (buy_price is null or buy_price >= 0);

-- 1b --------------------------------------------------------------- parts: price history
create table if not exists public.part_price_history (
  id uuid primary key default gen_random_uuid(),
  guide_id uuid not null references public.part_price_guide(id) on delete cascade,
  old_price_min numeric, old_price_max numeric, old_buy_price numeric,
  new_price_min numeric, new_price_max numeric, new_buy_price numeric,
  changed_by uuid references public.profiles(id),
  changed_at timestamptz not null default now()
);
create index if not exists idx_pph_guide on public.part_price_history (guide_id, changed_at desc);
alter table public.part_price_history enable row level security;
drop policy if exists "view price history" on public.part_price_history;
create policy "view price history" on public.part_price_history for select
  using (public.get_my_role() = any (array['admin','super_admin','comptoir','technicien']));

create or replace function public.log_price_change() returns trigger
  language plpgsql security definer set search_path to 'public' as $$
begin
  if new.price_min is distinct from old.price_min
     or new.price_max is distinct from old.price_max
     or new.buy_price is distinct from old.buy_price then
    insert into public.part_price_history
      (guide_id, old_price_min, old_price_max, old_buy_price,
       new_price_min, new_price_max, new_buy_price, changed_by)
    values (new.id, old.price_min, old.price_max, old.buy_price,
            new.price_min, new.price_max, new.buy_price, auth.uid());
  end if;
  return new;
end $$;
drop trigger if exists trg_log_price_change on public.part_price_guide;
create trigger trg_log_price_change after update on public.part_price_guide
  for each row execute function public.log_price_change();

-- 2 ---------------------------------------------------------------- phone models catalogue
create table if not exists public.phone_models (
  id uuid primary key default gen_random_uuid(),
  brand text not null,
  name text not null,
  release_year integer,
  screen_size numeric,
  screen_type text,
  battery_mah integer,
  charge_port text,
  chipset text,
  ram_gb text,
  storage_gb text,
  market_price_min numeric,
  market_price_max numeric,
  notes text,
  specs_verified boolean not null default false,
  created_by uuid default auth.uid() references public.profiles(id),
  updated_by uuid references public.profiles(id),
  updated_at timestamptz default now(),
  created_at timestamptz default now(),
  constraint phone_models_brand_check check (char_length(trim(both from brand)) > 0),
  constraint phone_models_name_check check (char_length(trim(both from name)) > 0),
  constraint phone_models_price_check check (
    market_price_min is null or (market_price_min > 0 and (market_price_max is null or market_price_max >= market_price_min)))
);
create unique index if not exists uq_phone_models_name on public.phone_models (lower(name));
create index if not exists idx_phone_models_brand on public.phone_models (lower(brand));

create or replace function public.touch_phone_models() returns trigger
  language plpgsql security definer set search_path to 'public' as $$
begin
  new.updated_at = now();
  new.updated_by = auth.uid();
  return new;
end $$;
drop trigger if exists trg_touch_phone_models on public.phone_models;
create trigger trg_touch_phone_models before insert or update on public.phone_models
  for each row execute function public.touch_phone_models();

import { useAuth } from '../../context/AuthContext'
import { t } from '../../i18n/LanguageContext'

const PORTS = ['USB-C', 'Lightning', 'Micro-USB']
const SCREEN_TYPES = ['IPS LCD', 'PLS LCD', 'AMOLED', 'Super AMOLED', 'Dynamic AMOLED', 'OLED']

function emptyForm() {
  return {
    brand: '', name: '', release_year: '', screen_size: '', screen_type: '', battery_mah: '',
    charge_port: 'USB-C', chipset: '', ram_gb: '', storage_gb: '',
    market_price_min: '', market_price_max: '', notes: '', specs_verified: false,
  }
}

const num = (v) => (v === '' || v == null ? null : Number(v))

function specLine(m) {
  return [
    m.screen_size ? m.screen_size + '"' : null,
    m.screen_type,
    m.battery_mah ? m.battery_mah + ' mAh' : null,
    m.charge_port,
  ].filter(Boolean).join(' · ')
}

function priceText(min, max) {
  if (min == null) return '—'
  const hasMax = max != null && Number(max) !== Number(min)
  return hasMax
    ? t('{price_min} - {price_max} DT', { price_min: Number(min).toFixed(2), price_max: Number(max).toFixed(2) })
    : t('{price_min} DT', { price_min: Number(min).toFixed(2) })
}

function SpecRow({ label, value }) {
  if (value == null || value === '') return null
  return (
    <div className="flex justify-between gap-3 py-1.5 border-b border-[#f0f0ea] last:border-b-0 text-sm">
      <span className="text-[#6b6b6b]">{label}</span>
      <span className="font-semibold text-[#1a1a1a] text-end">{value}</span>
    </div>
  )
}

export default function PhoneCatalog() {
  const { profile } = useAuth()
  const navigate = useNavigate()
  const isAdmin = ['admin', 'super_admin'].includes(profile?.role)

  const [models, setModels] = useState([])
  const [parts, setParts] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)
  const [search, setSearch] = useState('')
  const [brand, setBrand] = useState('')
  const [openId, setOpenId] = useState(null)

  const [showForm, setShowForm] = useState(false)
  const [editingId, setEditingId] = useState(null)
  const [form, setForm] = useState(emptyForm())
  const [saving, setSaving] = useState(false)
  const [formError, setFormError] = useState(null)

  useEffect(() => { load() }, [])

  async function load() {
    setLoading(true)
    setError(null)
    const [m, p] = await Promise.all([
      supabase.from('phone_models').select('*').order('brand', { ascending: true }).order('name', { ascending: true }),
      supabase.from('part_price_guide').select('id, device_model, part_type, quality_label, price_min, price_max, buy_price'),
    ])
    if (m.error) setError(m.error.message)
    else if (p.error) setError(p.error.message)
    setModels(m.data || [])
    setParts(p.data || [])
    setLoading(false)
  }

  // parts of each model: matched on the model name (not case sensitive)
  const partsByModel = useMemo(() => {
    const map = new Map()
    for (const r of parts) {
      const k = (r.device_model || '').trim().toLowerCase()
      if (!map.has(k)) map.set(k, [])
      map.get(k).push(r)
    }
    return map
  }, [parts])

  const brands = useMemo(() => [...new Set(models.map((m) => m.brand))].sort(), [models])

  const filtered = useMemo(() => {
    const q = search.trim().toLowerCase()
    return models.filter((m) => {
      const okText = !q || m.name.toLowerCase().includes(q) || m.brand.toLowerCase().includes(q) || (m.chipset || '').toLowerCase().includes(q)
      const okBrand = !brand || m.brand === brand
      return okText && okBrand
    })
  }, [models, search, brand])

  function openAdd() {
    setEditingId(null)
    setForm(emptyForm())
    setFormError(null)
    setShowForm(true)
  }

  function openEdit(m) {
    setEditingId(m.id)
    setForm({
      brand: m.brand || '', name: m.name || '', release_year: m.release_year ?? '', screen_size: m.screen_size ?? '',
      screen_type: m.screen_type || '', battery_mah: m.battery_mah ?? '', charge_port: m.charge_port || '',
      chipset: m.chipset || '', ram_gb: m.ram_gb || '', storage_gb: m.storage_gb || '',
      market_price_min: m.market_price_min ?? '', market_price_max: m.market_price_max ?? '',
      notes: m.notes || '', specs_verified: !!m.specs_verified,
    })
    setFormError(null)
    setShowForm(true)
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  async function submitForm(e) {
    e.preventDefault()
    setFormError(null)
    const min = num(form.market_price_min)
    const max = num(form.market_price_max)
    if (!form.brand.trim() || !form.name.trim()) {
      setFormError(t('Remplissez la marque et le nom du modèle'))
      return
    }
    if (min !== null && !(min > 0)) { setFormError(t('Le prix doit être supérieur à 0')); return }
    if (max !== null && min === null) { setFormError(t('Indiquez d\'abord le prix « de »')); return }
    if (max !== null && max < min) { setFormError(t('« À » doit être supérieur ou égal à « De »')); return }
    setSaving(true)
    const payload = {
      brand: form.brand.trim(),
      name: form.name.trim(),
      release_year: num(form.release_year),
      screen_size: num(form.screen_size),
      screen_type: form.screen_type.trim() || null,
      battery_mah: num(form.battery_mah),
      charge_port: form.charge_port || null,
      chipset: form.chipset.trim() || null,
      ram_gb: form.ram_gb.trim() || null,
      storage_gb: form.storage_gb.trim() || null,
      market_price_min: min,
      market_price_max: max,
      notes: form.notes.trim() || null,
      specs_verified: !!form.specs_verified,
    }
    const { error: err } = editingId
      ? await supabase.from('phone_models').update(payload).eq('id', editingId)
      : await supabase.from('phone_models').insert(payload)
    setSaving(false)
    if (err) {
      setFormError(err.code === '23505' ? t('Ce modèle existe déjà') : t('Erreur :') + ' ' + err.message)
      return
    }
    setShowForm(false)
    load()
  }

  async function remove(id) {
    if (!window.confirm(t('Supprimer ce modèle ?'))) return
    const { error: err } = await supabase.from('phone_models').delete().eq('id', id)
    if (err) { alert(t('Échec :') + ' ' + err.message); return }
    setShowForm(false)
    load()
  }

  const goParts = (m, add) =>
    navigate('/price-guide?model=' + encodeURIComponent(m.name) + (add ? '&add=1' : ''))

  const inp = 'w-full px-3.5 py-2.5 border border-[#e5e5e5] rounded-lg text-sm focus:outline-none focus:border-[#e4211b]'

  return (
    <div>
      <div className="flex flex-col gap-3 sm:flex-row sm:justify-between sm:items-end mb-6 pb-4 border-b border-[#e5e5e5]">
        <div>
          <h1 className="text-xl font-extrabold text-[#1a1a1a] mb-1">{t('Catalogue téléphones')}</h1>
          <p className="text-sm text-[#6b6b6b]">{t('Fiches techniques et prix des pièces par modèle')}</p>
        </div>
        <button onClick={showForm ? () => setShowForm(false) : openAdd}
          className="w-full sm:w-auto bg-[#1a1a1a] text-white font-semibold text-sm px-5 py-2.5 rounded-lg">
          {showForm ? t('Annuler') : t('+ Ajouter un modèle')}
        </button>
      </div>

      <div className="mb-5">
        <input value={search} onChange={(e) => setSearch(e.target.value)} className={inp + ' mb-3'}
          placeholder={t('Rechercher un modèle (ex. : A15, Redmi Note 12, iPhone 13)')} />
        {brands.length > 0 && (
          <div className="flex gap-2 overflow-x-auto -mx-1 px-1">
            <button onClick={() => setBrand('')}
              className={`shrink-0 text-xs px-3 py-1.5 rounded-lg border ${!brand ? 'bg-[#e4211b] text-white border-[#e4211b]' : 'border-[#e5e5e5] text-[#6b6b6b]'}`}>
              {t('Tout')}
            </button>
            {brands.map((b) => (
              <button key={b} onClick={() => setBrand(b === brand ? '' : b)}
                className={`shrink-0 text-xs px-3 py-1.5 rounded-lg border ${brand === b ? 'bg-[#e4211b] text-white border-[#e4211b]' : 'border-[#e5e5e5] text-[#6b6b6b]'}`}>
                {b}
              </button>
            ))}
          </div>
        )}
      </div>

      {error && <p className="bg-[#fdeaea] text-[#b3170f] text-sm rounded-lg px-4 py-3 mb-4">{error}</p>}

      {showForm && (
        <form onSubmit={submitForm} className="bg-white border border-[#e5e5e5] rounded-xl p-4 sm:p-6 mb-6 max-w-lg">
          <h3 className="font-bold text-sm mb-4">{editingId ? t('Modifier le modèle') : t('Nouveau modèle')}</h3>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-4">
            <div>
              <label className="block text-sm font-semibold mb-1.5">{t('Marque')}</label>
              <input className={inp} list="brand-options" placeholder="Samsung" value={form.brand}
                onChange={(e) => setForm({ ...form, brand: e.target.value })} />
              <datalist id="brand-options">{brands.map((b) => <option key={b} value={b} />)}</datalist>
            </div>
            <div>
              <label className="block text-sm font-semibold mb-1.5">{t('Nom du modèle')}</label>
              <input className={inp} placeholder="Samsung Galaxy A15" value={form.name}
                onChange={(e) => setForm({ ...form, name: e.target.value })} />
            </div>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-4">
            <div>
              <label className="block text-sm font-semibold mb-1.5">{t('Année')}</label>
              <input type="number" className={inp} value={form.release_year}
                onChange={(e) => setForm({ ...form, release_year: e.target.value })} />

            <div>

              <input type="number" step="0.01" className={inp} value={form.screen_size}

              <label className="block text-sm font-semibold mb-1.5">{t('RAM (Go)')}</label>
              <input className={inp} placeholder="4/6/8" value={form.ram_gb}
                onChange={(e) => setForm({ ...form, ram_gb: e.target.value })} />

              .sort((a, b) => a.part_type.localeCompare(b.part_type) || Number(a.price_min) - Number(b.price_min))
            return (
              <div key={m.id} className="bg-white border border-[#e5e5e5] rounded-xl overflow-hidden">

                    {mp.length > 0 && (
                      <span className="bg-[#eefaf1] text-[#1f8a4c] text-[11px] font-bold px-2 py-0.5 rounded-full">
                        {t('{n} pièces', { n: mp.length })}
                      </span>
                    )}
                    <span className="text-[#9a9a9a] text-xs">{isOpen ? '▲' : '▼'}</span>
                  </div>
                </button>

                {isOpen && (
                  <div className="px-4 pb-4 border-t border-[#f0f0ea]">
                    <div className="pt-2 mb-3">
                      <SpecRow label={t('Marque')} value={m.brand} />
                      <SpecRow label={t('Année')} value={m.release_year} />
                      <SpecRow label={t('Écran')} value={m.screen_size ? m.screen_size + '" ' + (m.screen_type || '') : m.screen_type} />
                      <SpecRow label={t('Batterie')} value={m.battery_mah ? m.battery_mah + ' mAh' : null} />
                      <SpecRow label={t('Port de charge')} value={m.charge_port} />
                      <SpecRow label={t('Processeur')} value={m.chipset} />
                      <SpecRow label={t('RAM (Go)')} value={m.ram_gb} />
                      <SpecRow label={t('Stockage (Go)')} value={m.storage_gb} />
                      <SpecRow label={t('Prix du téléphone')} value={m.market_price_min != null ? priceText(m.market_price_min, m.market_price_max) : null} />
                      <SpecRow label={t('Remarques')} value={m.notes} />
                    </div>
                    <p className={`text-xs mb-3 ${m.specs_verified ? 'text-[#1f8a4c]' : 'text-[#b36b00]'}`}>
                      {m.specs_verified ? t('✅ Fiche vérifiée') : t('⚠️ Fiche à vérifier')}
                    </p>

                    <h4 className="text-xs font-bold text-[#e4211b] mb-2">{t('Pièces et prix')}</h4>
                    {mp.length === 0 ? (
                      <p className="text-xs text-[#6b6b6b] mb-3">{t('Aucune pièce enregistrée pour ce modèle.')}</p>
                    ) : (
                      <div className="mb-3">
                        {mp.map((r) => (
                          <div key={r.id} className="flex justify-between items-center gap-3 py-1.5 border-b border-[#f0f0ea] last:border-b-0 text-sm">
                            <span className="min-w-0">
                              {r.part_type} — {r.quality_label}
                              {r.buy_price != null && (
                                <span className="block text-[11px] text-[#9a9a9a]">{t('Achat : {price} DT', { price: Number(r.buy_price).toFixed(2) })}</span>
                              )}
                            </span>
                            <span className="bg-[#eefaf1] text-[#1f8a4c] font-bold text-xs whitespace-nowrap px-2.5 py-1 rounded-full">
                              {priceText(r.price_min, r.price_max)}
                            </span>
                          </div>
                        ))}
                      </div>
                    )}

                    <div className="flex flex-wrap gap-2">
                      <button onClick={() => goParts(m, true)}
                        className="bg-[#e4211b] text-white text-xs font-semibold px-3.5 py-2 rounded-lg">
                        {t('+ Ajouter une pièce')}
                      </button>
                      <button onClick={() => goParts(m, false)}
                        className="text-xs text-[#6b6b6b] border border-[#e5e5e5] rounded-lg px-3.5 py-2">
                        {t('💲 Voir dans le guide des prix')}
                      </button>
                      <button onClick={() => openEdit(m)}
                        className="text-xs text-[#6b6b6b] border border-[#e5e5e5] rounded-lg px-3.5 py-2">
                        {t('✏️ Modifier la fiche')}
                      </button>
                    </div>
                  </div>
                )}
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}

src/lib/usePhoneModels.js

Helper: list of model names for the autocomplete.

import { useEffect, useState } from 'react'
import { supabase } from './supabaseClient'

// The names of the phone models in the catalogue (Catalogue téléphones).
// Used to autocomplete the "device model" field when receiving a device.
export function usePhoneModelNames() {
  const [names, setNames] = useState([])
  useEffect(() => {
    let alive = true
    supabase
      .from('phone_models')
      .select('name')
      .order('name')
      .then(({ data }) => {
        if (alive) setNames((data || []).map((r) => r.name))
      })
    return () => { alive = false }
  }, [])
  return names
}

// Common part types shown as quick filter chips + suggestions in the form.
// Free text is still allowed (e.g. "بورت شحن") — this list is just a shortcut.
const COMMON_PARTS = [t('Écran'), t('Batterie'), t('Écouteur'), t('Caméra'), t('Port de charge'), t('Vitre arrière')]
// part names are typed by the team, so the icon is found for the Arabic (old data), French and English words
const PART_ICONS = {
  'شاشة': '🖥️', 'بطارية': '🔋', 'سماعة': '🔊', 'كاميرا': '📷', 'بورت شحن': '🔌', 'زجاج خلفي': '🪟',
  'Écran': '🖥️', 'Batterie': '🔋', 'Écouteur': '🔊', 'Caméra': '📷', 'Port de charge': '🔌', 'Vitre arrière': '🪟',
  'Screen': '🖥️', 'Battery': '🔋', 'Earpiece': '🔊', 'Camera': '📷', 'Charging port': '🔌', 'Back glass': '🪟',
}
const partIcon = (type) => PART_ICONS[type] || '🔧'

// quality shortcuts: a screen can have 2, 3 or 4 qualities, each with its own price
const QUALITY_SUGGESTIONS = [t('Original'), t('OLED'), t('Copie 1er choix'), t('Copie')]
// prices change often: a price not touched for 7 days gets a "check again" badge
const STALE_DAYS = 7
const isStale = (r) => r.updated_at && Date.now() - new Date(r.updated_at).getTime() > STALE_DAYS * 86400000
const fmtP = (min, max) =>
  min == null ? '—' : max != null && Number(max) !== Number(min) ? Number(min).toFixed(2) + '-' + Number(max).toFixed(2) : Number(min).toFixed(2)

function priceLabel(r) {
  const hasMax = r.price_max != null && Number(r.price_max) !== Number(r.price_min)
  return hasMax
    ? t('{price_min} - {price_max} DT', { price_min: Number(r.price_min).toFixed(2), price_max: Number(r.price_max).toFixed(2) })
    : t('{price_min} DT', { price_min: Number(r.price_min).toFixed(2) })
}

  const { user, profile } = useAuth()
  // All three roles (admin, comptoir, technicien) can add / edit / delete prices.
  const canEdit = ['admin', 'super_admin', 'technicien', 'comptoir'].includes(profile?.role)
  const phoneModels = usePhoneModelNames()
  const [params] = useSearchParams()

  const [rows, setRows] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)
  const [search, setSearch] = useState(params.get('model') || '')
  const [partFilter, setPartFilter] = useState('')

  const [showForm, setShowForm] = useState(false)
  const [editingId, setEditingId] = useState(null)
  const [form, setForm] = useState(emptyForm())
  const [saving, setSaving] = useState(false)
  const [formError, setFormError] = useState(null)
  const [history, setHistory] = useState([])

  function emptyForm() {
    return { device_model: '', part_type: '', quality_label: '', price_min: '', price_max: '', buy_price: '', notes: '' }
  }

  useEffect(() => { load() }, [])

  // coming from the phone catalogue ("+ Ajouter une pièce"): open the form with the model filled in
  useEffect(() => {
    const m = params.get('model')
    if (m && params.get('add') === '1' && canEdit) {
      setEditingId(null)
      setForm({ ...emptyForm(), device_model: m })
      setShowForm(true)
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  async function load() {
    setLoading(true)
    setError(null)
    // two foreign keys to profiles (created_by / updated_by) -> name the one we want
    const { data, error: e } = await supabase
      .from('part_price_guide')
      .select(
        'id, device_model, part_type, quality_label, price_min, price_max, buy_price, notes, updated_at, ' +
        'profiles!part_price_guide_updated_by_fkey(full_name)'
      )
      .order('device_model', { ascending: true })
      .order('part_type', { ascending: true })
      .order('price_min', { ascending: true })
    if (e) setError(e.message)
    setRows(data || [])
    setLoading(false)
  }

  // Search box filters by device model OR part type OR quality label — all client-side,
  // since the whole guide is small enough to load once and filter instantly as you type.
  const filtered = useMemo(() => {
    const q = search.trim().toLowerCase()
    return rows.filter((r) => {
      const matchesText =
        !q ||
        r.device_model.toLowerCase().includes(q) ||
        r.part_type.toLowerCase().includes(q) ||
        r.quality_label.toLowerCase().includes(q)
      const matchesPart = !partFilter || r.part_type === partFilter
      return matchesText && matchesPart
    })
  }, [rows, search, partFilter])

  // group by device model so all rows for "iPhone 13" (screen x3, battery...) sit together
  const grouped = useMemo(() => {
    const map = new Map()
    for (const r of filtered) {
      if (!map.has(r.device_model)) map.set(r.device_model, [])
      map.get(r.device_model).push(r)
    }
    return [...map.entries()]
  }, [filtered])

  const partTypesInUse = useMemo(
    () => [...new Set(rows.map((r) => r.part_type))].sort(),
    [rows]
  )

  async function loadHistory(guideId) {
    const { data } = await supabase
      .from('part_price_history')
      .select('id, old_price_min, old_price_max, old_buy_price, new_price_min, new_price_max, new_buy_price, changed_at, profiles(full_name)')
      .eq('guide_id', guideId)
      .order('changed_at', { ascending: false })
      .limit(6)
    setHistory(data || [])
  }

  function openAdd() {
    setEditingId(null)
    setHistory([])
    setForm(emptyForm())
    setFormError(null)
    setShowForm(true)
  }

  function openEdit(r) {
    setEditingId(r.id)
    setForm({
      device_model: r.device_model, part_type: r.part_type, quality_label: r.quality_label,
      price_min: r.price_min, price_max: r.price_max ?? '', buy_price: r.buy_price ?? '', notes: r.notes || '',
    })
    setFormError(null)
    setHistory([])
    loadHistory(r.id)
    setShowForm(true)
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  async function submitForm(e) {
    e.preventDefault()
    setFormError(null)
    const min = Number(form.price_min)
    const max = form.price_max === '' ? null : Number(form.price_max)
    const buy = form.buy_price === '' ? null : Number(form.buy_price)
    if (!form.device_model.trim() || !form.part_type.trim() || !form.quality_label.trim() || min <= 0) {
      setFormError(t('Remplissez le modèle, le type de pièce, la qualité et au moins le prix « de »'))
      return
    }
    if (max !== null && max < min) {
      setFormError(t('« À » doit être supérieur ou égal à « De »'))
      return
    }
    if (buy !== null && !(buy >= 0)) {
      setFormError(t('Le prix d\'achat est invalide'))
      return
    }
    setSaving(true)
    const payload = {
      device_model: form.device_model.trim(),
      part_type: form.part_type.trim(),
      quality_label: form.quality_label.trim(),
      price_min: min,
      price_max: max,
      buy_price: buy,
      notes: form.notes.trim() || null,
    }
    const { error: err } = editingId
      ? await supabase.from('part_price_guide').update(payload).eq('id', editingId)
      : await supabase.from('part_price_guide').insert({ ...payload, created_by: user.id })
    setSaving(false)
    if (err) { setFormError(t('Erreur :') + ' ' + err.message); return }
    setShowForm(false)
    load()
  }

  async function remove(id) {
    if (!window.confirm(t('Supprimer définitivement ce prix ?'))) return
    const { error: err } = await supabase.from('part_price_guide').delete().eq('id', id)
    if (err) { alert(t('Échec :') + ' ' + err.message); return }
    load()
  }

            {t('Marge : {price} DT', { price: (Number(r.price_min) - Number(r.buy_price)).toFixed(2) })}
          </p>

        <p className="text-[10px] text-[#9a9a9a]">
          {t('Dernière modification')} : {r.profiles?.full_name || '—'} · {new Date(r.updated_at).toLocaleDateString(getLocale())}
          {isStale(r) && <span className="ms-1.5 bg-[#fff4e0] text-[#b36b00] font-semibold px-1.5 py-0.5 rounded">{t('⚠️ À revérifier')}</span>}
        </p>
      </div>
    )
  }

  return (
    <div>
      <div className="flex flex-col gap-3 sm:flex-row sm:justify-between sm:items-end mb-6 pb-4 border-b border-[#e5e5e5]">
        <div>
          <h1 className="text-xl font-extrabold text-[#1a1a1a] mb-1">{t('Guide des prix')}</h1>
          <p className="text-sm text-[#6b6b6b]">{t('Prix estimatifs (de - à) selon le modèle du téléphone et le type de pièce')}</p>
        </div>
        {canEdit && (
          <button onClick={showForm ? () => setShowForm(false) : openAdd}
            className="w-full sm:w-auto bg-[#1a1a1a] text-white font-semibold text-sm px-5 py-2.5 rounded-lg">
            {showForm ? t('Annuler') : t('+ Ajouter un prix')}
          </button>
        )}
      </div>

      {/* search + part-type chips */}
      <div className="mb-5">
        <input
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          placeholder={t('Rechercher par modèle de téléphone ou type de pièce (ex. : iPhone 13 ou écran)')}
          className={inp + ' mb-3'}
        />
        {partTypesInUse.length > 0 && (
          <div className="flex gap-2 overflow-x-auto -mx-1 px-1">
            <button onClick={() => setPartFilter('')}
              className={`shrink-0 text-xs px-3 py-1.5 rounded-lg border ${!partFilter ? 'bg-[#e4211b] text-white border-[#e4211b]' : 'border-[#e5e5e5] text-[#6b6b6b]'}`}>
              
              {t('Tout')}
            </button>
            {partTypesInUse.map((p) => (
              <button key={p} onClick={() => setPartFilter(p === partFilter ? '' : p)}
                className={`shrink-0 text-xs px-3 py-1.5 rounded-lg border flex items-center gap-1.5 ${partFilter === p ? 'bg-[#e4211b] text-white border-[#e4211b]' : 'border-[#e5e5e5] text-[#6b6b6b]'}`}>
                <span>{partIcon(p)}</span>{p}
              </button>
            ))}
          </div>
        )}
      </div>

      {error && <p className="bg-[#fdeaea] text-[#b3170f] text-sm rounded-lg px-4 py-3 mb-4">{error}</p>}

import arBase from './ar'

import arCatalog from './ar_catalog'
import { supabase } from '../lib/supabaseClient'

// Three languages: French (default), English and Arabic (right to left).
// The KEY of every text is its French wording, so:
//   t('Commandes')                       -> "Commandes" (fr)  /  "Orders" (en)
//   t('il y a {m} min', { m: 5 })        -> "il y a 5 min"    /  "5 min ago"
// A text with no English entry simply stays in French — it is never blank.
// If two French texts are identical but mean different things in English, add "|context"
// to the key (e.g. 'Reçu|reçu de caisse'): the part after | is never displayed.
//
// t() can be imported and used ANYWHERE (inside a component, in a helper, or in a list at the top
// of a file). Changing the language reloads the page once, so everything — including those lists
// at the top of files — is rebuilt in the new language. The choice is saved on the device.

const STORAGE_KEY = 'm4u-lang'
export const LANGUAGES = ['fr', 'en', 'ar']
// the catalogue texts are in their own files and merged here
const en = { ...enBase, ...enCatalog }
const ar = { ...arBase, ...arCatalog }
const DICTIONARIES = { fr: {}, en, ar }
const LOCALES = { fr: 'fr-FR', en: 'en-GB', ar: 'ar-TN' }

function readSavedLang() {
  try {
    const saved = localStorage.getItem(STORAGE_KEY)
    if (LANGUAGES.includes(saved)) return saved
  } catch { /* storage blocked: ignore */ }
  return 'fr' // French first, whatever the phone language is
}

// read once when the app loads (before any page code runs)
const current = readSavedLang()
// Arabic reads right to left: the whole page is mirrored (menu, tables, arrows). French and English: left to right.
const DIR = current === 'ar' ? 'rtl' : 'ltr'
if (typeof document !== 'undefined') {
  document.documentElement.lang = current
  document.documentElement.dir = DIR
}

export function getLang() { return current }
export function getLocale() { return LOCALES[current] }
export function getDir() { return DIR }

export function t(key, vars) {
  if (key == null) return ''
  const raw = String(key)
  const dict = DICTIONARIES[current] || {}
  let text = Object.prototype.hasOwnProperty.call(dict, raw) ? dict[raw] : raw
  // "Reçu|context" -> hide the context part when no translation exists
  if (text === raw && raw.includes('|')) text = raw.split('|')[0]
  if (vars) {
    text = text.replace(/\{(\w+)\}/g, (m, name) => (vars[name] !== undefined ? vars[name] : m))
  }
  return text
}

// Notifications are translated by the database (columns title_fr / title_en / message_fr / message_en).
// This picks the one for the language in use; old rows without a translation show the original text.
// Arabic is the text the database triggers write, so it is shown as it is.
export function notifTitle(n) {
  if (current === 'ar') return n.title || n.title_fr || ''
  return (current === 'en' ? n.title_en : n.title_fr) || n.title_fr || n.title || ''
}
export function notifMessage(n) {
  if (current === 'ar') return n.message || n.message_fr || ''
  return (current === 'en' ? n.message_en : n.message_fr) || n.message_fr || n.message || ''
}

// Remember the language in the person's profile too, so the PHONE notifications arrive in it.
export async function saveLangToProfile(lang) {
  try {
    await Promise.race([
      supabase.rpc('set_my_lang', { p_lang: lang }),
      new Promise((resolve) => setTimeout(resolve, 1500)),
    ])
  } catch { /* not signed in, or offline: the phone keeps its last language */ }
}

export async function setLang(next) {
  if (!LANGUAGES.includes(next) || next === current) return
  try { localStorage.setItem(STORAGE_KEY, next) } catch { /* ignore */ }
  await saveLangToProfile(next)
  window.location.reload()
}

const LanguageContext = createContext({ lang: 'fr', dir: 'ltr', setLang, t, locale: 'fr-FR' })

export function LanguageProvider({ children }) {
  useEffect(() => {
    document.documentElement.lang = current
    document.documentElement.dir = DIR
  }, [])
  const value = useMemo(() => ({ lang: current, dir: DIR, setLang, t, locale: LOCALES[current] }), [])
  return <LanguageContext.Provider value={value}>{children}</LanguageContext.Provider>
}

// const { t, lang, dir, locale, setLang } = useI18n()   (same t as the one you can import directly)
export function useI18n() {
  return useContext(LanguageContext)
}

                  className="w-full px-3 py-2 border border-[#e5e5e5] rounded-lg text-sm"
                  dir="ltr"
                />
              </div>
            </div>

            <h4 className="text-xs font-bold text-[#e4211b] mb-3">{t('Informations de l\'appareil')}</h4>
            <div className="mb-3">
              <label className="block text-xs text-[#6b6b6b] mb-1">{t('Modèle de l\'appareil')}</label>
              <datalist id="phone-model-options">{phoneModels.map((n) => <option key={n} value={n} />)}</datalist>
              <input
                list="phone-model-options"
                value={form.deviceModel}
                onChange={(e) => setForm({ ...form, deviceModel: e.target.value })}
                className="w-full px-3 py-2 border border-[#e5e5e5] rounded-lg text-sm"
              />
            </div>
            <div className="mb-3">
              <label className="block text-xs text-[#6b6b6b] mb-1">{t('Description de la panne')}</label>
              <textarea
                value={form.issueDescription}
                onChange={(e) => setForm({ ...form, issueDescription: e.target.value })}
                className="w-full px-3 py-2 border border-[#e5e5e5] rounded-lg text-sm min-h-[60px]"
              />
            </div>
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 mb-3">
              <div>
                <label className="block text-xs text-[#6b6b6b] mb-1">{t('De (DT)')}</label>
                <input
                  type="number"
                  value={form.priceMin}
                  onChange={(e) => setForm({ ...form, priceMin: e.target.value })}
                  className="w-full px-3 py-2 border border-[#e5e5e5] rounded-lg text-sm"
                />
              </div>
              <div>
                <label className="block text-xs text-[#6b6b6b] mb-1">{t('À (DT)')}</label>
                <input
                  type="number"
                  value={form.priceMax}
                  onChange={(e) => setForm({ ...form, priceMax: e.target.value })}
                  className="w-full px-3 py-2 border border-[#e5e5e5] rounded-lg text-sm"
                />
              </div>
            </div>
            <div className="mb-4">
              <label className="block text-xs text-[#6b6b6b] mb-1">{t('Technicien assigné')}</label>
              <select
                value={form.technicianId}
                onChange={(e) => setForm({ ...form, technicianId: e.target.value })}
                className="w-full px-3 py-2 border border-[#e5e5e5] rounded-lg text-sm"
              >
                <option value="">{t('-- Choisir un technicien --')}</option>
                {technicians.map((tc) => (
                  <option key={tc.id} value={tc.id}>{tc.full_name}{tc.id === user.id ? ' (' + t('Moi') + ')' : tc.role === 'admin' ? ' · Admin' : ''}</option>
                ))}
              </select>
            </div>

            {modalError && <p className="text-[#b3170f] text-sm mb-3">{modalError}</p>}

            <button
              onClick={handleSaveEdits}
              disabled={saving}
              className="w-full bg-[#1a1a1a] text-white font-semibold text-sm px-5 py-2.5 rounded-lg hover:opacity-90 disabled:opacity-60 mb-4"
            >
              {saving ? t('Enregistrement...') : t('Enregistrer les modifications')}
            </button>

            {/* fidele client: apply the discount? (comptoir or technicien can answer) */}
            <FideleDiscount order={selectedOrder} onChanged={handleDiscountChanged} />

            <div className="pt-4 border-t border-[#e5e5e5]">
              <h4 className="text-xs font-bold text-[#e4211b] mb-2">{t('Prix final (fixé par le technicien)')}</h4>
              {selectedOrder.final_price ? (
                <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 bg-[#faf9f5] rounded-lg p-3">
                  <div>
                    <span className="font-bold text-[#1a1a1a]">{selectedOrder.final_price}  {t('DT')}</span>
                    <span className={`text-xs me-2 ${selectedOrder.price_confirmed ? 'text-green-700' : 'text-[#b3170f]'}`}>
                      {selectedOrder.price_confirmed ? t('✔ Confirmé') : t('● Pas encore confirmé')}
                    </span>
                  </div>
                  {!selectedOrder.price_confirmed && (
                    <button
                      onClick={handleConfirmPrice}
                      disabled={confirming}
                      className="w-full sm:w-auto bg-green-700 text-white text-sm px-4 py-2 rounded-lg hover:opacity-90 disabled:opacity-60"
                    >
                      {confirming ? '...' : t('Confirmation du prix')}
                    </button>
                  )}
                </div>
              ) : (
                <p className="text-xs text-[#6b6b6b]">{t('Le technicien n\'a pas encore fixé le prix final.')}</p>
              )}

              {/* total / paid / remaining */}
              <PaymentSummary order={selectedOrder} />

              {/* QR: reprint the client receipt / device sticker */}
              <button
                type="button"
                onClick={() => setPrintOrder({
                  ...selectedOrder,
                  client_name: selectedOrder.clients?.full_name,
                })}
                className="w-full border border-[#e5e5e5] text-[#1a1a1a] text-sm font-semibold px-4 py-2.5 rounded-lg hover:bg-[#f7f7f7] mt-3"
              >
                
                {t('🖨 Imprimer le reçu / l\'étiquette (QR)')}
              </button>

              {/* Customer leaves without paying everything (only shows after the price is confirmed) */}
              {selectedOrder.status !== 'cancelled' && (
                <UnpaidDeliveryButton order={selectedOrder} onDone={handleUnpaidDone} />
              )}
            </div>

            <button
              type="button"
              onClick={closeModal}
              className="w-full border border-[#e5e5e5] text-[#1a1a1a] text-sm px-4 py-2.5 rounded-lg hover:bg-[#f7f7f7] mt-4"
            >
              
              {t('Fermer')}
            </button>
          </div>
        </div>
      )}

      {printOrder && <PrintOrderDialog order={printOrder} onClose={() => setPrintOrder(null)} />}
    </div>
  )
}
